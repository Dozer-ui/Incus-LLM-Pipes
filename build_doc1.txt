+------------------------------------------------------------------------------+
|                   Launch and Configure Base Build Container                  |
+------------------------------------------------------------------------------+
# Spin up a Debian 12 system container.

incus launch images:debian/12 llm-base

# Enter the container environment.

incus shell llm-base

# Update package index and install tools. Necessary tools: curl, jq, stdbuf, cat

apt-get update && apt-get install -y curl jq coreutils zstd

# Install the Ollama inference engine.

curl -fsSL https://ollama.com/install.sh | sh
     '----'                              '---'
       |                                   |
       |                                   |

       |                                     into "sh" which executes the 
       |                                     install scripts directly, not
       |                                     writing to disk. Not best practice
       |                                     in most environments but okay for
       |                                     this disposable container.
       |

       + -f fail silently, no error outputs
       + -s silent, mutes progress meter and default error messages
       + -S used with -s to show error only if connection fails completely
       + -L follows HTTP redirects if server forwards request to new URL

# Create configuration directory. Establishes a standard filesystem path to hold node-specific system prompts.
Node specific system prompt will live here in system_prompt.txt

mkdir -p /etc/llm

# Create the stream wrapper on host. 
Captures upstream stdin, prepends the local /etc/llm/system_prompt.txt, and uses stdbuf to force unbuffered token streaming to stdout.

+------------------------------------------------------------------------------+
|                                  llm-node                                    |
+------------------------------------------------------------------------------+
#!/usr/bin/env bash

# Read system prompt if file exists, default to empty string
SYSTEM_PROMPT=""
if [ -f /etc/llm/system_prompt.txt ]; then
    SYSTEM_PROMPT=$(cat /etc/llm/system_prompt.txt)
fi

# Read raw incoming byte stream from stdin
USER_INPUT=$(cat -)

# Prevent execution if stdin is completely empty
if [ -z "$USER_INPUT" ]; then
    exit 0
fi

# Combine system prompt + user payload and stream token output
# Replace 'qwen2.5:3b' with your target model choice
stdbuf -oL -o0 ollama run qwen2.5:3b "$SYSTEM_PROMPT

---
User Input:
$USER_INPUT"
+------------------------------------------------------------------------------+

# Create file above on host named llm-node. Use Incus to push to container.

incus file push llm-node llm-base/usr/local/bin/llm-node

# Go back inside the container and make file executable.

chmod +x /usr/local/bin/llm-node

# Exit container shell sesion.

exit

# On host stop container.

incus stop llm-base

# Publish container as local image template.
Converts configured container into Incus image alias. You can now use this image to spin up preconfigured llm-nodes for llm prompt chaining infrastructure.

incus publish llm-base --alias llm-node-base description="Debian 12 base image with Ollama and llm-node wrapper"

# Clean up. Removes temporary build container.

incus delete llm-base

# Checklist:
[x] Preconfigured Incus image for LLM prompt chaining.
